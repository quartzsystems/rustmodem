# +MS, +ES and +DS take set, read and test forms. +MS is read and is
# reported in full; +ES and +DS are stored and have no effect yet. Extended
# commands are separated by ; and a basic command may follow one.
> ATE0\r
< ATE0\r\r\nOK\r\n
> AT+MS=B103,0,300,300,300,300\r
< \r\nOK\r\n
> AT+MS?\r
< \r\n+MS: B103,0,300,300,300,300\r\n\r\nOK\r\n
> AT+ES=3,0,2;+DS=3,0,2048,32;S0=1\r
< \r\nOK\r\n
! s 0=1

< \r\n+ES: 3,0,2\r\n\r\n+DS: 3,0,2048,32\r\n\r\nOK\r\n
